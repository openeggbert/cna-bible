% Cross-reference macros used throughout the book. \cnaclass and \cnans double as the book's
% index markup: every class/method/namespace name they wrap is indexed automatically at every
% occurrence, so the back-of-book index is populated directly from the same markup that already
% exists throughout the text, rather than requiring a second, separate pass of manual \index{}
% placement.
\newcommand{\cnaclass}[1]{\texttt{#1}\index{#1}}
\newcommand{\cnans}[1]{\texttt{#1}\index{#1}}

  cna public/.style={cna node, fill=diagrampublic, very thick},
  cna internal/.style={cna node, fill=diagraminternal},

  cna warning/.style={cna node, fill=diagramwarning, dashed},
